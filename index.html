async def main(wf): angles = ['Market overview...', 'Technical...', 'Expert opinions...', ...] market_data = await wf.map_agents( items=angles, subagent_type='web_searcher', max_concurrency=3 ) verified_data = await wf.map_agents( items=market_data, subagent_type='fact_checker' ) final_report = await wf.reduce_agent( items=verified_data, subagent_type='synthesizer' ) return final_report
